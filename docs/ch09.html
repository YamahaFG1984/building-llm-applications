第 9 章

问题改写

上一章改进的是「资料怎么存」，这一章改进的是「拿什么去找」。 用户随口一问的句子，往往并不适合直接拿去做语义检索。办法是：先让 LLM 把问题加工一下，再去检索。 五种加工方式，对应五种不同的病症。

本章目标

  • 理解为什么「用户的原话」常常不是好的检索查询
  • 掌握五种问题改写技巧：查询重写、多查询、后退一步、HyDE、问题分解
  • 看懂它们共同的代码骨架 —— 改的只是「拿去检索的那句话」
  • 知道每种技巧适合什么场景、代价是什么

为什么要改写问题

语义检索比较的是「问题的向量」和「文本块的向量」。但这两样东西天生不太像：

  • 问题是疑问句，资料是陈述句；
  • 问题里常有「帮我」「请告诉我」「好玩的」这类口语化、含糊的词，它们在资料里并不出现，却会把向量「带偏」；
  • 一个问题可能太具体（资料里没有一模一样的细节），也可能太笼统，或者一句话里塞了好几个问题。

看一个真实例子。知识库是英国旅游目的地的 Wikivoyage 页面（按 h2 小节切块），直接用原话检索：

user_question = "Tell me some fun things I can enjoy in Cornwall"
initial_results = uk_granular_collection.similarity_search(query=user_question, k=4)
# 结果：命中的多是泛泛的介绍或不太相关的小节 ——「fun things I can enjoy」太含糊了

共同的骨架

这一章的五种技巧，在代码结构上几乎一模一样。标准 RAG 链里，问题同时走两条路：一条去检索，一条进提示词。 问题改写只动第一条路 —— 进提示词让模型作答的，始终是用户的原始问题。

先把共用的部分准备好：模型、检索器、RAG 提示词。后面每种技巧只需要再加一条「改写链」。

from langchain_openai import ChatOpenAI
from langchain_core.output_parsers import StrOutputParser
from langchain_core.prompts import ChatPromptTemplate
from langchain_core.runnables import RunnablePassthrough

llm = ChatOpenAI(model="gpt-5-nano", openai_api_key=OPENAI_API_KEY)
retriever = uk_granular_collection.as_retriever()

rag_prompt = ChatPromptTemplate.from_template("""
Given a question and some context, answer the question.
If you do not know the answer, just say I do not know.

Context: {context}
Question: {question}
""")

def make_rag_chain(search_chain):
    """search_chain：输入原始问题，输出检索到的文档。五种技巧只是换这个参数。"""
    return (
        {"context": search_chain, "question": RunnablePassthrough()}
        | rag_prompt | llm | StrOutputParser())

import { ChatOpenAI } from "@langchain/openai";
import { StringOutputParser } from "@langchain/core/output_parsers";
import { ChatPromptTemplate } from "@langchain/core/prompts";
import { Runnable, RunnablePassthrough, RunnableSequence } from "@langchain/core/runnables";
import type { Document } from "@langchain/core/documents";

const llm = new ChatOpenAI({ model: "gpt-5-nano" });
const retriever = ukGranularCollection.asRetriever();
const formatDocs = (docs: Document[]) => docs.map((d) => d.pageContent).join("\n\n");

const ragPrompt = ChatPromptTemplate.fromTemplate(`
Given a question and some context, answer the question.
If you do not know the answer, just say I do not know.

Context: {context}
Question: {question}
`);

/** searchChain：输入原始问题，输出检索到的文档。五种技巧只是换这个参数。 */
function makeRagChain(searchChain: Runnable<string, Document[]>) {
  return RunnableSequence.from([
    { context: searchChain.pipe(formatDocs), question: new RunnablePassthrough() },
    ragPrompt, llm, new StringOutputParser(),
  ]);
}

// 所有「让 LLM 加工问题」的提示词，集中声明在一个 AI Service 接口里（方法随各小节逐个添加）。
interface QuestionTransformer {
    @UserMessage("""
            Given a question and some context, answer the question.
            If you do not know the answer, just say I do not know.

            Context: {{context}}
            Question: {{question}}""")
    String answer(@V("context") String context, @V("question") String question);
}

QuestionTransformer ai = AiServices.create(QuestionTransformer.class, llm);

/** 用给定的查询去检索，返回拼接好的上下文 */
static String retrieve(String searchQuery, int k) {
    var request = EmbeddingSearchRequest.builder()
            .queryEmbedding(embeddingModel.embed(searchQuery).content())
            .maxResults(k).build();
    return ukGranularStore.search(request).matches().stream()
            .map(m -> m.embedded().text())
            .collect(Collectors.joining("\n\n"));

① 查询重写（Rewrite-Retrieve-Read）

ⓘ 对症：问题口语化、啰嗦、关键词不突出

让 LLM 把用户的话改写成一条简洁、关键词明确、适合向量检索的查询。

② 多查询（Multi-Query）

ⓘ 对症：一种问法只能覆盖一个角度；或者一个问题里其实包含好几个独立的小问题

让 LLM 从不同角度生成多条查询，分别检索，再把结果合并、去重。检索的「网」撒得更大。

from typing import List
from langchain_core.output_parsers import BaseOutputParser
from langchain_classic.retrievers.multi_query import MultiQueryRetriever

multi_query_gen_prompt = ChatPromptTemplate.from_template("""
You are an AI language model assistant. Your task is to generate five different versions
of the given user question to retrieve relevant documents from a vector database.
By generating multiple perspectives on the user question, your goal is to help the user
overcome some of the limitations of the distance-based similarity search.
Provide these alternative questions separated by newlines.
Original question: {question}
""")

class LineListOutputParser(BaseOutputParser[List[str]]):
    """自定义解析器：模型每行输出一个问题，按行拆成列表。"""
    def parse(self, text: str) -> List[str]:
        return list(filter(None, text.strip().split("\n")))

multi_query_gen_chain = multi_query_gen_prompt | llm | LineListOutputParser()

# 方式一：把自己的查询生成链交给 MultiQueryRetriever
multi_query_retriever = MultiQueryRetriever(
    retriever=retriever, llm_chain=multi_query_gen_chain, parser_key="lines")

# 方式二：全部用默认配置，一行搞定
std_multi_query_retriever = MultiQueryRetriever.from_llm(retriever=retriever, llm=llm)

# 它本身就是一个检索器（内部已完成：生成多条查询 → 分别检索 → 合并去重）
multi_query_rag_chain = make_rag_chain(multi_query_retriever)
print(multi_query_rag_chain.invoke("Tell me some fun things I can do in Cornwall"))

import { RunnableLambda } from "@langchain/core/runnables";

// 方式一：全部用默认配置，一行搞定
const stdMultiQueryRetriever = MultiQueryRetriever.fromLLM({
  llm, retriever, queryCount: 5 });

// 方式二：自己写，看清它内部做了什么
const multiQueryGenPrompt = ChatPromptTemplate.fromTemplate(`
You are an AI language model assistant. Your task is to generate five different versions
of the given user question to retrieve relevant documents from a vector database.
By generating multiple perspectives on the user question, your goal is to help the user
overcome some of the limitations of the distance-based similarity search.
Provide these alternative questions separated by newlines.
Original question: {question}
`);

const multiQueryGenChain = multiQueryGenPrompt
  .pipe(llm).pipe(new StringOutputParser())
  .pipe((text: string) => text.trim().split("\n").filter(Boolean));   // 按行拆成列表

const customMultiQueryRetriever = RunnableLambda.from(async (question: string) => {
  const queries = await multiQueryGenChain.invoke({ question });
  const groups = await Promise.all(queries.map((q) => retriever.invoke(q)));   // 并行检索
  const unique = new Map(groups.flat().map((d) => [d.pageContent, d]));        // 按内容去重
  return [...unique.values()];
});

const multiQueryRagChain = makeRagChain(customMultiQueryRetriever);
console.log(await multiQueryRagChain.invoke("Tell me some fun things I can do in Cornwall"));

💡 这也叫「单步分解」

如果原问题是「康沃尔有什么好玩的，住哪里方便，怎么过去？」，生成的多条查询就自然地把它拆成了几个独立的子问题。 只要子问题之间互不依赖、可以并行检索，这就够用了。互相依赖的情况见本章最后一节。

③ 后退一步（Step-Back）

ⓘ 对症：问题太具体，资料里找不到一模一样的细节

让 LLM 把问题抽象一层，问一个更宽泛的问题去检索，拿回更完整的背景知识。

具体的问题

Can you give me some tips for a trip to Brighton?
资料里没有哪一段叫「tips」

后退一步的问题

What should I know before traveling to a UK seaside city?
能带回交通、住宿、安全、景点等一大片背景

llm = ChatOpenAI(model="gpt-5", openai_api_key=OPENAI_API_KEY)   # 「抽象」需要更强一点的模型

step_back_prompt = ChatPromptTemplate.from_template("""

进阶做法是两个都搜：用原始问题检索一次（拿细节），用后退问题再检索一次（拿背景），把两份结果都放进上下文。

④ 假设文档嵌入（HyDE）

ⓘ 对症：「问句」和「陈述句」长得不像，语义距离偏大

先让 LLM 凭空写一个可能的答案（内容对不对无所谓），再用这段「假答案」去检索。 因为假答案和真资料都是陈述句、用词风格相近，向量会靠得更近。 

hyde_prompt = ChatPromptTemplate.from_template("""

Do not add anything else.

Sentence:
""")

hyde_chain = hyde_prompt | llm | StrOutputParser()

print(hyde_chain.invoke("What are the best beaches in Cornwall?"))

hyde_rag_chain = make_rag_chain(
    {"question": RunnablePassthrough()} | hyde_chain | retriever)

⚠ 假答案只用来检索，千万别进提示词

假设文档是模型编的，很可能有错。它的唯一作用是当检索的「诱饵」。 最终作答时，上下文必须是检索回来的真实资料 —— 而且 RAG 提示词最好像上面的仓库代码那样加一句 Only use the provided context to answer the question。 另外，如果问题涉及模型完全不了解的领域（比如你公司的内部项目代号），它写出的假答案会离题万里，HyDE 反而帮倒忙。

⑤ 多步分解

ⓘ 对症：问题由几个有先后依赖的小问题组成，前一个的答案是后一个的前提

例如：「康沃尔最受欢迎的沙滩，八月的平均气温是多少？」要回答它，必须先知道最受欢迎的沙滩是哪个， 才能去查那个地方的气温。多查询的并行拆分在这里不管用。

生成计划的提示词：

Break down the following question into multiple dependent steps that I can execute on a
vector store or other data sources (e.g., databases) to obtain the final answer.
Assume you have granular data in your data sources, but not aggregate data.
Each subsequent question can take a parameter from the result of the previous step.
For each step, include the question and the corresponding parameter that will be filled
with the previous answer.
Provide the sequence of questions in JSON format as a list, where each step includes:
- Question: The text of the question.
- Parameter: The parameter to be filled with the previous answer.
----
Original question: {user_question}
Multiple dependent questions:

原书没有给出完整实现，下面是一个最小的执行循环，帮助你理解它的运作方式：

from pydantic import BaseModel, Field

class Step(BaseModel):
    question: str = Field(description="The question for this step. Use [PREVIOUS] where "
                                      "the previous step's answer must be inserted.")

class Plan(BaseModel):
    steps: List[Step]

planner = ChatPromptTemplate.from_template(
    "Break down the following question into multiple dependent steps ...\n"
    "Original question: {user_question}"
) | llm.with_structured_output(Plan)

step_rag_chain = make_rag_chain(retriever)     # 每一步都是一次普通的 RAG

def multi_step_answer(user_question: str) -> str:
    plan = planner.invoke({"user_question": user_question})
    previous_answer, collected = "", []

    for step in plan.steps:
        question = step.question.replace("[PREVIOUS]", previous_answer)   # 代入上一步答案
        previous_answer = step_rag_chain.invoke(question)
        collected.append(f"Q: {question}\nA: {previous_answer}")

    # 最后：用收集到的全部中间结论回答原始问题
    final_prompt = rag_prompt.format(context="\n\n".join(collected), question=user_question)
    return llm.invoke(final_prompt).content

import { z } from "zod";

const Plan = z.object({
  steps: z.array(z.object({
    question: z.string().describe(
      "The question for this step. Use [PREVIOUS] where the previous step's answer must be inserted."),
  })),
});

const planner = ChatPromptTemplate.fromTemplate(
  "Break down the following question into multiple dependent steps ...\n" +
  "Original question: {user_question}",
).pipe(llm.withStructuredOutput(Plan));

const stepRagChain = makeRagChain(retriever);     // 每一步都是一次普通的 RAG

async function multiStepAnswer(userQuestion: string): Promise<string> {
  const plan = await planner.invoke({ user_question: userQuestion });
  let previousAnswer = "";
  const collected: string[] = [];

  for (const step of plan.steps) {
    const question = step.question.replace("[PREVIOUS]", previousAnswer);   // 代入上一步答案
    previousAnswer = await stepRagChain.invoke(question);
    collected.push(`Q: ${question}\nA: ${previousAnswer}`);
  }

  // 最后：用收集到的全部中间结论回答原始问题
  const finalPrompt = await ragPrompt.format({
    context: collected.join("\n\n"), question: userQuestion });
  return (await llm.invoke(finalPrompt)).content as string;
}

record Step(String question) {}

// 在 QuestionTransformer 接口里添加：
@UserMessage("""
        Break down the following question into multiple dependent steps ...
        Use [PREVIOUS] where the previous step's answer must be inserted.
        Original question: {{user_question}}""")
List<Step> plan(@V("user_question") String userQuestion);

static String multiStepAnswer(QuestionTransformer ai, String userQuestion) {
    String previousAnswer = "";
    List<String> collected = new ArrayList<>();

    for (Step step : ai.plan(userQuestion)) {
        String question = step.question().replace("[PREVIOUS]", previousAnswer);  // 代入上一步答案
        previousAnswer = ai.answer(retrieve(question, 4), question);              // 一次普通的 RAG
        collected.add("Q: " + question + "\nA: " + previousAnswer);
    }
    // 最后：用收集到的全部中间结论回答原始问题
    return ai.answer(String.join("\n\n", collected), userQuestion);
}
🔬 这已经是智能体的雏形

「LLM 订计划 → 程序逐步执行 → 把结果反馈回去」，正是第 11 章要讲的智能体循环的简化版。 区别在于这里的计划是一次性订好的；真正的智能体每一步都会根据最新结果重新决定下一步， 而且每一步可以选用不同的工具（向量库、SQL、API……）。

五种技巧怎么选

技巧拿去检索的是对症额外开销
查询重写精炼后的查询口语化、啰嗦、关键词不突出1 次 LLM
多查询多个角度的问题一种问法覆盖不全；含多个独立子问题1 次 LLM + N 次检索
后退一步更抽象的问题问题过于具体，需要背景知识1 次 LLM（可能要更强的模型）
HyDE一段假设的答案问句与陈述句的语义落差1 次 LLM
多步分解依次执行的子问题子问题之间有先后依赖多次 LLM + 多次检索
🔑 每种技巧都在用时间和钱换准确率

它们都至少多一次模型调用，用户会多等一会儿。不要默认全部开启： 用 LangSmith 看看检索到底差在哪里，只加能解决那个具体问题的技巧。 改写这一步对模型能力要求不高，通常用最小、最快的模型就够了。

小结

  • 用户的原话常常不适合直接做语义检索：太口语、太具体、太笼统，或一句话里有多个问题。
  • 问题改写只改变「拿去检索的那句话」，最终让模型作答的仍是原始问题。
  • 查询重写：提炼关键词。多查询：多角度撒网，合并去重。后退一步：抽象一层拿背景。
  • HyDE：用一段假设的答案去检索，因为「答案像答案」。假答案绝不能进最终提示词。
  • 多步分解：先订计划，逐步执行，把上一步答案代入下一步 —— 已经接近智能体了。
  • 所有技巧都有成本，先诊断，后用药。

动手练习

  1. 对 "Tell me some fun things I can enjoy in Cornwall" 分别用原始问题和重写后的查询检索，把两组结果并排比较。
  2. 打印多查询生成的 5 个问题。有没有哪个角度是你自己想不到的？
  3. 实现后退一步的「两个都搜」版本：原始问题和后退问题各检索一次，结果合并后再回答。
  4. 给 HyDE 一个模型不可能知道的问题（比如一个虚构地名的景点），看它生成的假设文档是什么，检索结果如何。
  5. 把第 7 章「对话记忆」里那个追问失败的问题（"And then what did they do?"）用查询重写解决：让改写链同时看到对话历史。
